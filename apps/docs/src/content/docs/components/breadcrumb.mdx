import DemoFrame from '~/components/DemoFrame.astro';
import BasicVue from '~/components/demos/breadcrumb/BasicVue.vue';
import SeparatorsVue from '~/components/demos/breadcrumb/SeparatorsVue.vue';
import LongVue from '~/components/demos/breadcrumb/LongVue.vue';

## 基础用法

`items` 数组每项 `{ label, href? }`。最后一项不论是否带 `href` 都被渲染成当前页（不可点）。

<DemoFrame
  vueCode={`const items = [
  { label: '首页', href: '/' },
  { label: '组件', href: '/components/' },
  { label: 'Breadcrumb 面包屑' },
];

<CfBreadcrumb :items="items" />`}
  reactCode={`const items = [
  { label: '首页', href: '/' },
  { label: '组件', href: '/components/' },
  { label: 'Breadcrumb 面包屑' },
];

<CfBreadcrumb items={items} />`}
>
  <BasicVue client:load />
</DemoFrame>

## 自定义分隔符

`separator` 默认 `/`，常用 `›` / `·` / `→`。任何字符都行。

<DemoFrame
  vueCode={`<CfBreadcrumb :items="items" separator="/" />
<CfBreadcrumb :items="items" separator="›" />
<CfBreadcrumb :items="items" separator="·" />
<CfBreadcrumb :items="items" separator="→" />`}
  reactCode={`<CfBreadcrumb items={items} separator="/" />
<CfBreadcrumb items={items} separator="›" />
<CfBreadcrumb items={items} separator="·" />
<CfBreadcrumb items={items} separator="→" />`}
>
  <SeparatorsVue client:load />
</DemoFrame>

## 长链路

层级较深时，前几项变成可点击链接，最后一项是当前位置。

<DemoFrame
  vueCode={`const items = [
  { label: '首页', href: '/' },
  { label: '组织', href: '/orgs/' },
  { label: '某团队', href: '/orgs/team-x/' },
  { label: '项目', href: '/orgs/team-x/projects/' },
  { label: '深层目录' },
];

<CfBreadcrumb :items="items" separator="›" />`}
  reactCode={`<CfBreadcrumb items={items} separator="›" />`}
>
  <LongVue client:load />
</DemoFrame>

## API

| Prop | 类型 | 默认值 | 说明 |
|---|---|---|---|
| `items` | `Array<{ label: string; href?: string }>` | `[]` | 链路项；最后一项作为当前页 |
| `separator` | `string` | `'/'` | 分隔符；常用 `›` `·` 或自定义 |
