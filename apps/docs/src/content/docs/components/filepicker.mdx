import DemoFrame from '~/components/DemoFrame.astro';
import BasicVue from '~/components/demos/filepicker/BasicVue.vue';

## 基础用法

`multiple` 多选，`accept` MIME / 后缀过滤；选中文件以 chip 形式列在按钮旁，单个删除或一键清空。

<DemoFrame
  vueCode={`<CfFilePicker v-model="files" multiple accept="image/*,application/json" />`}
  reactCode={`<CfFilePicker value={files} onChange={setFiles} multiple accept="image/*" />`}
>
  <BasicVue client:load />
</DemoFrame>

## API

| 属性 | 类型 | 默认值 | 说明 |
|---|---|---|---|
| `modelValue` / `value` | `File[]` | `[]` | |
| `multiple` | `boolean` | `false` | |
| `accept` | `string` | — | 同原生 `<input type=file>` |
| `size` | `'sm' \| 'md' \| 'lg'` | `'md'` | |
| `variant` | `'outline' \| 'plain'` | `'outline'` | |
| `disabled` / `showFiles` | `boolean` | `false` / `true` | |
| `buttonText` / `emptyText` | `string` | `'选择文件'` / `'未选择文件'` | |
