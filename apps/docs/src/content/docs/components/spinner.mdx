import DemoFrame from '~/components/DemoFrame.astro';
import BasicVue from '~/components/demos/spinner/BasicVue.vue';
import SizesVue from '~/components/demos/spinner/SizesVue.vue';
import TonesVue from '~/components/demos/spinner/TonesVue.vue';
import LabelVue from '~/components/demos/spinner/LabelVue.vue';

## 基础用法

无 prop 渲染默认尺寸 + 主色的旋转环。

<DemoFrame
  vueCode={`<CfSpinner />`}
  reactCode={`<CfSpinner />`}
>
  <BasicVue client:load />
</DemoFrame>

## 三档尺寸

`size` —— `sm` 行内文字大小 / `md` 默认 / `lg` 模态级别加载提示。

<DemoFrame
  vueCode={`<CfSpinner size="sm" />
<CfSpinner size="md" />
<CfSpinner size="lg" />`}
  reactCode={`<CfSpinner size="sm" />
<CfSpinner size="md" />
<CfSpinner size="lg" />`}
>
  <SizesVue client:load />
</DemoFrame>

## 6 种 tone

`tone` 控制描边颜色 — `primary`（默认）/ `success` / `warning` / `danger` / `info` / `neutral`。语义化对齐 token 里 `--accent-*` / `--status-*` / `--fg-*`。

<DemoFrame
  vueCode={`<CfSpinner tone="primary" />
<CfSpinner tone="success" />
<CfSpinner tone="warning" />
<CfSpinner tone="danger" />
<CfSpinner tone="info" />
<CfSpinner tone="neutral" />`}
  reactCode={`<CfSpinner tone="primary" />
<CfSpinner tone="success" />
<CfSpinner tone="warning" />
<CfSpinner tone="danger" />
<CfSpinner tone="info" />
<CfSpinner tone="neutral" />`}
>
  <TonesVue client:load />
</DemoFrame>

## 文字标签

`label` 在右侧加可读文案，同时设为 `aria-label`，对屏读友好。

<DemoFrame
  vueCode={`<CfSpinner label="正在加载…" />
<CfSpinner size="lg" tone="success" label="保存中" />`}
  reactCode={`<CfSpinner label="正在加载…" />
<CfSpinner size="lg" tone="success" label="保存中" />`}
>
  <LabelVue client:load />
</DemoFrame>

## API

| Prop | 类型 | 默认值 | 说明 |
|---|---|---|---|
| `size` | `'sm' \| 'md' \| 'lg'` | `'md'` | 尺寸 |
| `tone` | `'primary' \| 'success' \| 'warning' \| 'danger' \| 'info' \| 'neutral'` | `'primary'` | 颜色 |
| `label` | `string` | — | 文字标签 + aria-label |
